// rejudge 用 DeepSeek 重新评审 G2 基线（原由 gpt-4o 评审），再与本次 memeval 结果做配对精确符号检验。
// 用法：cp 本文件 /tmp/rejudge/main.go，在 goagent 根目录执行 go run /tmp/rejudge/main.go <参数>。
package main

)

// judgeRetries 是首次失败后的最大重试次数。
const judgeRetries = 4

type key struct {
	system, question string
	run              int
}

func keyOf(r memeval.Result) key { return key{r.System, r.QuestionID, r.Run} }

// signP 是精确双侧符号检验：p = min(1, 2·Σ_{i≤min(w,l)} C(n,i)/2^n)。
// 用大整数求和，n 很大时也不会溢出。
func signP(w, l int) float64 {
	n := w + l
	sum := new(big.Int)
	for i := 0; i <= min(w, l); i++ {
		sum.Add(sum, new(big.Int).Binomial(int64(n), int64(i)))
	}
	p, _ := new(big.Rat).SetFrac(sum.Lsh(sum, 1), new(big.Int).Lsh(big.NewInt(1), uint(n))).Float64()
	return math.Min(1, p)
}

func check() error {
	cases := []struct {
		name string
		got  float64
		want float64
	}{
		{"signP(19,24) 对应 G2 README 的 0.542", math.Round(signP(19, 24)*1000) / 1000, 0.542},
		{"signP(0,0)", signP(0, 0), 1},
		{"signP(200,200) 不溢出", signP(200, 200), 1},
		{"signP(0,10)", signP(0, 10), 2.0 / 1024},
	}
	for _, c := range cases {
		if math.Abs(c.got-c.want) > 1e-12 {
			return fmt.Errorf("%s：得到 %v，期望 %v", c.name, c.got, c.want)
		}
	}
	return nil
}

func readResults(path string) ([]memeval.Result, error) {
	f, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer f.Close()
	var out []memeval.Result
	sc := bufio.NewScanner(f)
	sc.Buffer(make([]byte, 1<<20), 64<<20)
	for sc.Scan() {
		if len(bytes.TrimSpace(sc.Bytes())) == 0 {
			continue
		}
		var r memeval.Result
		if err := json.Unmarshal(sc.Bytes(), &r); err != nil {
			return nil, fmt.Errorf("%s：%w", path, err)
		}
		out = append(out, r)
	}
	return out, sc.Err()
}

// readBaseline 按文件顺序给 run 赋值（两个基线文件内部的 run 字段都是 1，不可信）。
func readBaseline(paths []string) ([]memeval.Result, error) {
	var all []memeval.Result
	for i, p := range paths {
		rs, err := readResults(p)
		if err != nil {
			return nil, err
		}
		for j := range rs {
			rs[j].Run = i + 1
		}
		all = append(all, rs...)
	}
	return all, nil
}

// writeResultsAtomic 先写同目录临时文件再改名，崩溃时不会留下半截文件。
func writeResultsAtomic(path string, rs []memeval.Result) error {
	tmp := path + ".tmp"
	if err := writeResults(tmp, rs); err != nil {
		return err
	}
	return os.Rename(tmp, path)
}

func writeResults(path string, rs []memeval.Result) error {
	var buf bytes.Buffer
	for _, r := range rs {
		b, err := json.Marshal(r)
		if err != nil {
			return err
		}
		buf.Write(b)
		buf.WriteByte('\n')
	}
	return os.WriteFile(path, buf.Bytes(), 0o644)
}

// rejudge 用 judge 重判全部基线；仍失败的题不进结果，只计数。
func rejudge(ctx context.Context, judge llm.Client, base []memeval.Result, concurrency int) (judged []memeval.Result, failed int) {
	ctx, abort := context.WithCancel(ctx)
	defer abort()
	out := make([]*memeval.Result, len(base))
	sem := make(chan struct{}, max(concurrency, 1))
	var wg sync.WaitGroup
	for i := range base {
		wg.Add(1)
		sem <- struct{}{}
		go func() {
			defer wg.Done()
			defer func() { <-sem }()
			r := base[i]
			q := memeval.Question{Text: r.Question, Answer: r.Answer, Type: r.Type, Abstain: r.Abstain}
			var (
				resp *llm.CompletionResponse
				err  error
			)
			for attempt := 0; attempt <= judgeRetries; attempt++ {
				if resp, err = llm.Chat(ctx, judge, []llm.Message{llm.UserMessage(memeval.JudgePrompt(q, r.Hypothesis))}); err == nil {
					break
				}
				// 鉴权或模型不存在时每题重试也无意义，直接终止整轮，避免对每题白等退避。
				if reason := llm.ClassifyError(err).Reason; reason == llm.ReasonAuth || reason == llm.ReasonNotFound {
					abort()
					return
				}
				if !common.IsRetryable(err) {
					break
				}
				select {
				case <-ctx.Done():
					return
				case <-time.After(time.Duration(attempt+1) * 2 * time.Second):
				}
			}
			if err != nil {
				return
			}
			r.Correct = strings.Contains(strings.ToLower(resp.Content), "yes")
			out[i] = &r
		}()
	}
	wg.Wait()
	for _, r := range out {
		if r == nil {
			failed++
			continue
		}
		judged = append(judged, *r)
	}
	return judged, failed
}

type acc struct{ right, total int }

func (a acc) String() string {
	if a.total == 0 {
		return "-"
	}
	return fmt.Sprintf("%.1f%% (%d/%d)", 100*float64(a.right)/float64(a.total), a.right, a.total)
}

func accuracy(rs []memeval.Result) map[[2]any]acc {
	m := map[[2]any]acc{}
	for _, r := range rs {
		k := [2]any{r.System, r.Run}
		a := m[k]
		a.total++
		if r.Correct {
			a.right++
		}
		m[k] = a
	}
	return m
}

func index(rs []memeval.Result) map[key]memeval.Result {
	m := make(map[key]memeval.Result, len(rs))
	for _, r := range rs {
		m[keyOf(r)] = r
	}
	return m
}

func render(orig, judged, cur []memeval.Result, failed int) string {
	jIdx, oIdx := index(judged), index(orig)
	curAcc, jAcc, oAcc := accuracy(cur), accuracy(judged), accuracy(orig)

	type cell struct{ w, l int }
	cells := map[[2]any]cell{}
	paired, agree, both := 0, 0, 0
	for _, c := range cur {
		k := keyOf(c)
		b, ok := jIdx[k]
		if !ok {
			continue
		}
		paired++
		ck := [2]any{c.System, c.Run}
		x := cells[ck]
		switch {
		case c.Correct && !b.Correct:
			x.w++
		case !c.Correct && b.Correct:
			x.l++
		}
		cells[ck] = x
	}
	for k, j := range jIdx {
		if o, ok := oIdx[k]; ok {
			both++
			if o.Correct == j.Correct {
				agree++
			}
		}
	}

	keys := make([][2]any, 0, len(curAcc))
	for k := range curAcc {
		keys = append(keys, k)
	}
	slices.SortFunc(keys, func(a, b [2]any) int {
		return cmp.Or(cmp.Compare(a[0].(string), b[0].(string)), cmp.Compare(a[1].(int), b[1].(int)))
	})

	var sb strings.Builder
	sb.WriteString("# 跨运行对比（配对精确符号检验）\n\n")
	sb.WriteString("| 方案 | 运行 | 本次准确率 | 基线(DeepSeek 重判)准确率 | 基线(gpt-4o 原判)准确率 | 胜 | 负 | p |\n|---|---|---|---|---|---|---|---|\n")
	for _, k := range keys {
		c := cells[k]
		fmt.Fprintf(&sb, "| %s | %d | %s | %s | %s | %d | %d | %.3f |\n", k[0], k[1], curAcc[k], jAcc[k], oAcc[k], c.w, c.l, signP(c.w, c.l))
	}
	sb.WriteString("\n")
	fmt.Fprintf(&sb, "配对题数 n = %d（键为 方案+题目+运行；胜/负之外为双方判定相同）。\n\n", paired)
	if both > 0 {
		fmt.Fprintf(&sb, "DeepSeek 与 gpt-4o 判定一致率：%.1f%%（%d/%d）。\n\n", 100*float64(agree)/float64(both), agree, both)
	} else {
		sb.WriteString("DeepSeek 与 gpt-4o 判定一致率：无可比条目。\n\n")
	}
	fmt.Fprintf(&sb, "重判失败并剔除出配对的基线题数：%d。\n", failed)
	return sb.String()
}

func main() {
	var (
		baseline    = flag.String("baseline", "", "G2 基线文件，逗号分隔，按顺序对应 run 1、run 2")
		current     = flag.String("current", "", "本次 memeval 结果 jsonl")
		judged      = flag.String("judged", "", "重新评审后的基线文件；已存在时直接读取，不调用模型")
		out         = flag.String("out", "compare.md", "对比结果 Markdown")
		concurrency = flag.Int("concurrency", 8, "重新评审并发数")
		doCheck     = flag.Bool("check", false, "只验证符号检验")
	)
	flag.Parse()
	if err := run(*baseline, *current, *judged, *out, *concurrency, *doCheck); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func run(baseline, current, judgedPath, out string, concurrency int, doCheck bool) error {
	if doCheck {
		if err := check(); err != nil {
			return err
		}
		fmt.Println("ok")
		return nil
	}
	if baseline == "" || current == "" || judgedPath == "" {
		return errors.New("需要 -baseline、-current 与 -judged")
	}
	orig, err := readBaseline(strings.Split(baseline, ","))
	if err != nil {
		return err
	}
	cur, err := readResults(current)
	if err != nil {
		return err
	}

	var rejudged []memeval.Result
	failed := 0
	if _, statErr := os.Stat(judgedPath); statErr == nil {
		if rejudged, err = readResults(judgedPath); err != nil {
			return err
		}
		failed = len(orig) - len(rejudged)
	} else {
		// 温度 0 与 memeval 命令行的判定客户端一致，密钥由环境变量提供。
		judge, err := providers.NewDeepSeekWithOptions(llm.WithTemperature(0))
		if err != nil {
			return err
		}
		ctx, stop := signalContext()
		defer stop()
		rejudged, failed = rejudge(ctx, judge, orig, concurrency)
		// 中断或有失败时不落盘，否则下次会把不完整文件当作完整的重判结果。
		if failed == 0 && ctx.Err() == nil {
			if err := writeResultsAtomic(judgedPath, rejudged); err != nil {
				return err
			}
		} else {
			fmt.Fprintf(os.Stderr, "重判失败 %d 题，未写入 %s，重跑会重新评审全部题目\n", failed, judgedPath)
		}
	}
	return os.WriteFile(out, []byte(render(orig, rejudged, cur, max(failed, 0))), 0o644)
}

func signalContext() (context.Context, context.CancelFunc) {
	return signal.NotifyContext(context.Background(), os.Interrupt)
}
