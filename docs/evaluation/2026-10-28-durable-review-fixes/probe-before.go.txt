type model struct{ calls int; label string }
func(m *model)Provider() constants.Provider{return constants.Provider("fake")}
func(m *model)IsAvailable()bool{return true}
func(m *model)Complete(context.Context,*llm.CompletionRequest)(*llm.CompletionResponse,error){panic("unused")}
func(m *model)Chat(context.Context,[]llm.Message)(*llm.CompletionResponse,error){panic("unused")}
func(m *model)ChatStream(ctx context.Context,msg []llm.Message)(<-chan *llm.StreamChunk,error){return m.CompleteStream(ctx,&llm.CompletionRequest{Messages:msg})}
func(m *model)CompleteStream(context.Context,*llm.CompletionRequest)(<-chan *llm.StreamChunk,error){m.calls++;ch:=make(chan *llm.StreamChunk,1);ch<-&llm.StreamChunk{Content:m.label,Delta:m.label,Done:true,FinishReason:"stop",Usage:&llm.Usage{TotalTokens:10}};close(ch);return ch,nil}
func(m *model)CompleteToolsStream(ctx context.Context,r *llm.CompletionRequest,_ []interfaces.Tool)(<-chan *llm.StreamChunk,error){return m.CompleteStream(ctx,r)}
type legacyModel struct{ llm.StreamClient; selections int }
func(m *legacyModel)GenerateWithTools(context.Context,string,[]interfaces.Tool)(*common.ToolCallResponse,error){m.selections++;if m.selections%2==1{return &common.ToolCallResponse{ToolCalls:[]common.ToolCall{{Name:"write",Arguments:map[string]interface{}{}}}},nil};return &common.ToolCallResponse{},nil}
func check(e error){if e!=nil{panic(e)}}
func store() durable.Store{dir,e:=os.MkdirTemp("","goagent-review-store-");check(e);s,e:=durable.NewFileStore(dir);check(e);return s}
