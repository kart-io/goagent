# ACP Agent 端实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 goagent-contrib 新增独立模块 `acp`，让任意 builder 构建的 Agent 以 Agent Client Protocol Agent 身份经 stdio 被编辑器驱动，并在 goagent-examples 提供可运行案例。

**Architecture:** `Server.Serve` 为每条连接创建 SDK 的 `AgentSideConnection`，连接对象 `conn` 实现 SDK 的 `acpsdk.Agent`/`acpsdk.AgentLoader`。每个 ACP 会话对应一个 `Session`（工作区、Pipeline、env 实现、未结束调用表）和宿主 Factory 构建的 `Runner`。`session/prompt` 把 `ExecuteStream` 的文本事件转为 `session/update`，工具通知与审批由 `Session.Pipeline()` 的钩子发出，文件与命令按客户端能力转发给编辑器或回退本地。

**Tech Stack:** Go 1.25、`github.com/coder/acp-go-sdk` v0.13.5、goagent v0.32.0（builder、tools、core/session、harness/env、llm）、testify。

**Spec:** `docs/superpowers/specs/2026-10-31-acp-agent-design.md`（goagent 仓库）

```go
// fakeEditor 是测试中的 ACP 客户端：记录收到的通知与请求，按配置回答审批、文件与终端请求。
type fakeEditor struct {
	mu          sync.Mutex
	updates     []acpsdk.SessionNotification
	permissions []acpsdk.RequestPermissionRequest
	// answer 为空时选择 allow_once。
	answer func(context.Context, acpsdk.RequestPermissionRequest) acpsdk.RequestPermissionResponse
	files  map[string]string // 编辑器缓冲区：绝对路径 -> 内容
	writes []acpsdk.WriteTextFileRequest
	reads  []string
	// terminal 决定终端命令的输出与退出码；hang 为真时 wait_for_exit 阻塞到请求取消。
	terminal  func(acpsdk.CreateTerminalRequest) (output string, exit int)
	hang      bool
	created   []acpsdk.CreateTerminalRequest
	termCalls []string
}

func newEditor() *fakeEditor { return &fakeEditor{files: map[string]string{}} }

var _ acpsdk.Client = (*fakeEditor)(nil)

func (e *fakeEditor) SessionUpdate(_ context.Context, n acpsdk.SessionNotification) error {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.updates = append(e.updates, n)
	return nil
}

func (e *fakeEditor) RequestPermission(ctx context.Context, r acpsdk.RequestPermissionRequest) (acpsdk.RequestPermissionResponse, error) {
	e.mu.Lock()
	e.permissions = append(e.permissions, r)
	answer := e.answer
	e.mu.Unlock()
	if answer == nil {
		return acpsdk.RequestPermissionResponse{Outcome: acpsdk.NewRequestPermissionOutcomeSelected("allow_once")}, nil
	}
	return answer(ctx, r), nil
}

func (e *fakeEditor) ReadTextFile(_ context.Context, r acpsdk.ReadTextFileRequest) (acpsdk.ReadTextFileResponse, error) {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.reads = append(e.reads, r.Path)
	content, ok := e.files[r.Path]
	if !ok {
		return acpsdk.ReadTextFileResponse{}, acpsdk.NewInternalError(map[string]any{"error": "no buffer"})
	}
	return acpsdk.ReadTextFileResponse{Content: content}, nil
}

func (e *fakeEditor) WriteTextFile(_ context.Context, r acpsdk.WriteTextFileRequest) (acpsdk.WriteTextFileResponse, error) {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.writes = append(e.writes, r)
	e.files[r.Path] = r.Content
	return acpsdk.WriteTextFileResponse{}, nil
}

func (e *fakeEditor) CreateTerminal(_ context.Context, r acpsdk.CreateTerminalRequest) (acpsdk.CreateTerminalResponse, error) {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.created = append(e.created, r)
	e.termCalls = append(e.termCalls, "create")
	return acpsdk.CreateTerminalResponse{TerminalId: "term-1"}, nil
}

func (e *fakeEditor) result() (string, int) {
	e.mu.Lock()
	defer e.mu.Unlock()
	if e.terminal == nil || len(e.created) == 0 {
		return "", 0
	}
	return e.terminal(e.created[len(e.created)-1])
}

func (e *fakeEditor) record(call string) {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.termCalls = append(e.termCalls, call)
}

func (e *fakeEditor) WaitForTerminalExit(ctx context.Context, _ acpsdk.WaitForTerminalExitRequest) (acpsdk.WaitForTerminalExitResponse, error) {
	e.record("wait")
	e.mu.Lock()
	hang := e.hang
	e.mu.Unlock()
	if hang {
		<-ctx.Done()
		return acpsdk.WaitForTerminalExitResponse{}, ctx.Err()
	}
	_, code := e.result()
	return acpsdk.WaitForTerminalExitResponse{ExitCode: acpsdk.Ptr(code)}, nil
}

func (e *fakeEditor) TerminalOutput(context.Context, acpsdk.TerminalOutputRequest) (acpsdk.TerminalOutputResponse, error) {
	e.record("output")
	out, _ := e.result()
	return acpsdk.TerminalOutputResponse{Output: out}, nil
}

func (e *fakeEditor) KillTerminal(context.Context, acpsdk.KillTerminalRequest) (acpsdk.KillTerminalResponse, error) {
	e.record("kill")
	return acpsdk.KillTerminalResponse{}, nil
}

func (e *fakeEditor) ReleaseTerminal(context.Context, acpsdk.ReleaseTerminalRequest) (acpsdk.ReleaseTerminalResponse, error) {
	e.record("release")
	return acpsdk.ReleaseTerminalResponse{}, nil
}

func (e *fakeEditor) notifications() []acpsdk.SessionNotification {
	e.mu.Lock()
	defer e.mu.Unlock()
	return slices.Clone(e.updates)
}

// agentText 拼接全部 agent_message_chunk 的文本。
func (e *fakeEditor) agentText() string {
	var out string
	for _, n := range e.notifications() {
		if c := n.Update.AgentMessageChunk; c != nil && c.Content.Text != nil {
			out += c.Content.Text.Text
		}
	}
	return out
}

// start 用两对 io.Pipe 连接 Server 与 fakeEditor，测试结束时断开并等待 Serve 返回。
func start(t *testing.T, srv *Server, ed *fakeEditor) *harness {
	t.Helper()
	agentIn, editorOut := io.Pipe()
	editorIn, agentOut := io.Pipe()
	ctx, cancel := context.WithCancel(context.Background())
	done := make(chan error, 1)
	go func() { done <- srv.Serve(ctx, agentIn, agentOut) }()
	h := &harness{editor: ed, client: acpsdk.NewClientSideConnection(ed, editorOut, editorIn)}
	t.Cleanup(func() {
		_ = editorOut.Close()
		_ = agentOut.Close()
		cancel()
		<-done
	})
	return h
}

func (h *harness) initialize(t *testing.T, caps acpsdk.ClientCapabilities) acpsdk.InitializeResponse {
	t.Helper()
	resp, err := h.client.Initialize(context.Background(), acpsdk.InitializeRequest{ProtocolVersion: acpsdk.ProtocolVersionNumber, ClientCapabilities: caps})
	require.NoError(t, err)
	return resp
}

	t.Helper()
	resp, err := h.client.NewSession(context.Background(), acpsdk.NewSessionRequest{Cwd: cwd, McpServers: []acpsdk.McpServer{}})

// requireCode 断言 err 是指定码的 JSON-RPC 错误。
func requireCode(t *testing.T, err error, code int) {
	t.Helper()
	var re *acpsdk.RequestError
	require.True(t, errors.As(err, &re), "期望 JSON-RPC 错误，得到 %v", err)
	require.Equal(t, code, re.Code, "%v", err)
}

// turn 是一次模型请求的结果：parts 逐块作为增量发出；wait 为真时阻塞到 ctx 结束。
type turn struct {
	parts []string
	calls []llm.ToolCall
	wait  bool
}

func say(parts ...string) turn { return turn{parts: parts} }

func calls(names ...string) turn {
	var cs []llm.ToolCall
	for i, name := range names {
		c := llm.ToolCall{ID: "c" + string(rune('1'+i)), Type: "function"}
		c.Function.Name = name
		c.Function.Arguments = `{"q":"A1"}`
		cs = append(cs, c)
	}
	return turn{calls: cs}
}

func waitTurn() turn { return turn{wait: true} }

// scriptModel 按顺序给出回合并记录请求；started 在每次请求开始时收到一个信号。
type scriptModel struct {
	mu       sync.Mutex
	turns    []turn
	requests []*llm.CompletionRequest
	started  chan struct{}
}

func newModel(turns ...turn) *scriptModel {
	return &scriptModel{turns: turns, started: make(chan struct{}, 16)}
}

func (m *scriptModel) Provider() constants.Provider { return constants.ProviderCustom }
func (m *scriptModel) IsAvailable() bool            { return true }
func (m *scriptModel) Complete(context.Context, *llm.CompletionRequest) (*llm.CompletionResponse, error) {
	return nil, errors.New("测试模型只支持流式")
}
func (m *scriptModel) Chat(context.Context, []llm.Message) (*llm.CompletionResponse, error) {
	return nil, errors.New("测试模型只支持流式")
}
func (m *scriptModel) CompleteStream(ctx context.Context, req *llm.CompletionRequest) (<-chan *llm.StreamChunk, error) {
	return m.CompleteToolsStream(ctx, req, nil)
}
func (m *scriptModel) ChatStream(ctx context.Context, ms []llm.Message) (<-chan *llm.StreamChunk, error) {
	return m.CompleteToolsStream(ctx, &llm.CompletionRequest{Messages: ms}, nil)
}
func (m *scriptModel) CompleteToolsStream(ctx context.Context, req *llm.CompletionRequest, _ []interfaces.Tool) (<-chan *llm.StreamChunk, error) {
	m.mu.Lock()
	copied := *req
	copied.Messages = slices.Clone(req.Messages)
	m.requests = append(m.requests, &copied)
	i := len(m.requests) - 1
	m.mu.Unlock()
	m.started <- struct{}{}
	if i >= len(m.turns) {
		return nil, errors.New("发生了多余的模型请求")
	}
	t := m.turns[i]
	ch := make(chan *llm.StreamChunk, len(t.parts)+1)
	if t.wait {
		go func() {
			<-ctx.Done()
			ch <- &llm.StreamChunk{Error: ctx.Err()}
			close(ch)
		}()
		return ch, nil
	}
	var content string
	for _, p := range t.parts {
		content += p
		ch <- &llm.StreamChunk{Delta: p, Content: content}
	}
	last := &llm.StreamChunk{Content: content, Done: true, ToolCalls: t.calls, FinishReason: "stop", Usage: &llm.Usage{TotalTokens: 2}}
	if len(t.calls) > 0 {
		last.FinishReason = "tool_calls"
	}
	ch <- last
	close(ch)
	return ch, nil
}

func (m *scriptModel) request(i int) *llm.CompletionRequest {
	m.mu.Lock()
	defer m.mu.Unlock()
	return m.requests[i]
}

func (m *scriptModel) count() int {
	m.mu.Lock()
	defer m.mu.Unlock()
	return len(m.requests)
}

// plainFactory 构建不带工具的 Agent，并把收到的会话交给 seen（可为 nil）。
func plainFactory(model llm.Client, seen chan<- *Session) Factory {
	return func(_ context.Context, s *Session) (Runner, error) {
		if seen != nil {
			seen <- s
		}
		agent, err := builder.NewSimpleBuilder(model).WithSessionID(s.ID()).WithMaxIterations(4).Build()
		if err != nil {
			return nil, err
		}
		return agent, nil
	}
}
```

```go
func TestInitializeReportsCapabilities(t *testing.T) {
	h := start(t, NewServer(plainFactory(newModel(), nil), WithAgentInfo("demo", "1.2.3")), newEditor())
	resp := h.initialize(t, acpsdk.ClientCapabilities{})
	require.EqualValues(t, acpsdk.ProtocolVersionNumber, resp.ProtocolVersion)

	require.Equal(t, "1.2.3", resp.AgentInfo.Version)
	require.False(t, resp.AgentCapabilities.LoadSession, "未配置会话存储时不支持 session/load")
	require.Nil(t, resp.AgentCapabilities.SessionCapabilities.List)
	require.NotNil(t, resp.AgentCapabilities.SessionCapabilities.Close)
	require.True(t, resp.AgentCapabilities.PromptCapabilities.EmbeddedContext)
	require.False(t, resp.AgentCapabilities.PromptCapabilities.Image)

	h2 := start(t, NewServer(plainFactory(newModel(), nil), WithSessionStore(session.NewMemoryStore())), newEditor())
	resp = h2.initialize(t, acpsdk.ClientCapabilities{})
	require.True(t, resp.AgentCapabilities.LoadSession)
	require.NotNil(t, resp.AgentCapabilities.SessionCapabilities.List)
}

func TestNewSessionBuildsAgentPerSession(t *testing.T) {
	seen := make(chan *Session, 2)
	h := start(t, NewServer(plainFactory(newModel(), seen)), newEditor())
	h.initialize(t, acpsdk.ClientCapabilities{})
	dir := t.TempDir()
	require.NoError(t, os.WriteFile(filepath.Join(dir, "a.txt"), []byte("磁盘内容"), 0o644))

	id1 := h.newSession(t, dir)
	id2 := h.newSession(t, dir)
	require.NotEqual(t, id1, id2)
	s := <-seen
	require.Equal(t, string(id1), s.ID())
	require.Equal(t, dir, s.Cwd())
	require.Empty(t, s.MCPServers())
	<-seen

	// 客户端未声明 fs 与 terminal 能力时使用本地实现。
	data, _, err := s.FileSystem().ReadFile(context.Background(), "a.txt")
	require.NoError(t, err)
	require.Equal(t, "磁盘内容", string(data))
	require.False(t, s.Executor().Capabilities().Remote)
```

```go
func newSession(c *conn, id, cwd string, mcp []acpsdk.McpServer) (*Session, error) {
	if !filepath.IsAbs(cwd) {
		return nil, fmt.Errorf("acp: cwd 必须是绝对路径：%q", cwd)
	}
	local, err := env.NewLocalFS(cwd)
	if err != nil {
		return nil, fmt.Errorf("acp: 打开工作区 %q：%w", cwd, err)
	}
	s := &Session{id: id, cwd: cwd, mcp: mcp, conn: c, local: local}
	s.fs = local
	s.exec = env.LocalExecutor{Root: cwd}
	return s, nil
}
```

```go
// Factory 为每个 ACP 会话构建一个 Agent。构建时必须 WithSessionID(s.ID())，工具经 s.Pipeline()
// 所在的 tools.Dispatcher 注册；配置了会话存储时还须 WithSessionLog 使用同一个 Store。
type Factory func(ctx context.Context, s *Session) (Runner, error)

// Runner 是本包对 Agent 的最小要求，*builder.ConfigurableAgent[C, S] 满足它。
type Runner interface {
	ExecuteStream(ctx context.Context, input any) (<-chan builder.AgentEvent, error)
}

// Option 配置 Server。
type Option func(*Server)

// WithAgentInfo 设置 initialize 回报的 Agent 名称与版本，默认 goagent、dev。
func WithAgentInfo(name, version string) Option {
	return func(s *Server) { s.name, s.version = name, version }
}

// WithSessionStore 开启 session/load 与 session/list；Factory 须以同一 Store 调用 WithSessionLog。
func WithSessionStore(store session.Store) Option {
	return func(s *Server) { s.store = store }
}

// WithToolKind 决定工具调用通知中的 kind，默认全部为 other。
func WithToolKind(kind func(tool string) acpsdk.ToolKind) Option {
	return func(s *Server) { s.kind = kind }
}

// Server 把 Factory 构建的 Agent 暴露为 ACP Agent。
type Server struct {
	factory       Factory
	name, version string
	store         session.Store
	kind          func(string) acpsdk.ToolKind
}

// NewServer 创建 Server。
func NewServer(f Factory, opts ...Option) *Server {
	s := &Server{factory: f, name: "goagent", version: "dev", kind: func(string) acpsdk.ToolKind { return acpsdk.ToolKindOther }}
	for _, opt := range opts {
		opt(s)
	}
	return s
}

// Serve 处理一条连接，直到对端断开（返回 nil）或 ctx 结束（返回 ctx.Err()）。返回时取消全部进行中的 prompt；
// SDK 的读循环在 in 关闭后才退出，ctx 结束后调用方应关闭 in。
func (s *Server) Serve(ctx context.Context, in io.Reader, out io.Writer) error {
	if s.factory == nil {
		return errors.New("acp: Factory 不能为空")
	}
	base, cancel := context.WithCancel(ctx)
	defer cancel()
	c := &conn{srv: s, base: base, ready: make(chan struct{}), sessions: map[string]*Session{}, cwds: map[string]string{}}
	c.peer = acpsdk.NewAgentSideConnection(c, out, in)
	close(c.ready)
	select {
	case <-c.peer.Done():
		return nil
	case <-ctx.Done():
		return ctx.Err()
	}
}

// conn 是一条 ACP 连接上的 Agent 实现。
type conn struct {
	srv   *Server
	base  context.Context
	peer  *acpsdk.AgentSideConnection
	ready chan struct{} // peer 赋值后关闭，SDK 处理请求的 goroutine 经 client() 读取 peer

	mu       sync.Mutex
	caps     acpsdk.ClientCapabilities
	sessions map[string]*Session
	cwds     map[string]string // 本连接新建或加载过的会话的工作区，供 session/list
}

var (
	_ acpsdk.Agent       = (*conn)(nil)
	_ acpsdk.AgentLoader = (*conn)(nil)
)

func (c *conn) client() *acpsdk.AgentSideConnection {
	<-c.ready
	return c.peer
}

func (c *conn) capabilities() acpsdk.ClientCapabilities {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.caps
}

func internalError(op string, err error) error {
	return acpsdk.NewInternalError(map[string]any{"error": fmt.Sprintf("acp: %s：%v", op, err)})
}

func (c *conn) Initialize(_ context.Context, p acpsdk.InitializeRequest) (acpsdk.InitializeResponse, error) {
	c.mu.Lock()
	c.caps = p.ClientCapabilities
	c.mu.Unlock()
	caps := acpsdk.AgentCapabilities{
		LoadSession:         c.srv.store != nil,
		PromptCapabilities:  acpsdk.PromptCapabilities{EmbeddedContext: true},
		SessionCapabilities: acpsdk.SessionCapabilities{Close: &acpsdk.SessionCloseCapabilities{}},
	}
	if c.srv.store != nil {
		caps.SessionCapabilities.List = &acpsdk.SessionListCapabilities{}
	}
	return acpsdk.InitializeResponse{
		ProtocolVersion:   acpsdk.ProtocolVersionNumber,
		AgentCapabilities: caps,
		AgentInfo:         &acpsdk.Implementation{Name: c.srv.name, Version: c.srv.version},
		AuthMethods:       []acpsdk.AuthMethod{},
	}, nil
}

// open 构造会话并调用 Factory，成功后登记；同 ID 的旧会话被替换。
func (c *conn) open(ctx context.Context, id, cwd string, mcp []acpsdk.McpServer) (*Session, error) {
	s, err := newSession(c, id, cwd, mcp)
	if err != nil {
		return nil, acpsdk.NewInvalidParams(map[string]any{"error": err.Error()})
	}
	runner, err := c.srv.factory(ctx, s)
	if err != nil {
		return nil, internalError("构建 Agent", err)
	}
	if runner == nil {
		return nil, internalError("构建 Agent", errors.New("Factory 返回了 nil Runner"))
	}
	s.runner = runner
	c.mu.Lock()
	c.sessions[id] = s
	c.cwds[id] = cwd
	c.mu.Unlock()
	return s, nil
}

func (c *conn) session(id acpsdk.SessionId) (*Session, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	s, ok := c.sessions[string(id)]
	if !ok {
		return nil, acpsdk.NewInvalidParams(map[string]any{"error": fmt.Sprintf("acp: 会话 %q 不存在", id)})
	}
	return s, nil
}

	if _, err := c.session(p.SessionId); err != nil {

}
```

```go
package acp

import (
	"context"
	"errors"
	"fmt"
	"strings"

	acpsdk "github.com/coder/acp-go-sdk"
)

// Prompt 执行一轮对话。ctx 被取消（session/cancel、同会话的新 prompt、连接断开）时返回 cancelled 而非错误，这是协议要求。
func (c *conn) Prompt(ctx context.Context, p acpsdk.PromptRequest) (acpsdk.PromptResponse, error) {
	s, err := c.session(p.SessionId)
	if err != nil {
		return acpsdk.PromptResponse{}, err
	}
	input, err := promptText(p.Prompt)
	if err != nil {
		return acpsdk.PromptResponse{}, acpsdk.NewInvalidParams(map[string]any{"error": err.Error()})
	}
	ctx, cancel := context.WithCancel(ctx)
	defer cancel()
	defer context.AfterFunc(c.base, cancel)()

	s.run.Lock()
	defer s.run.Unlock()
	cancelled := acpsdk.PromptResponse{StopReason: acpsdk.StopReasonCancelled}
	if ctx.Err() != nil {
		return cancelled, nil
	}
	events, err := s.runner.ExecuteStream(ctx, input)
	if err != nil {
		if ctx.Err() != nil {
			return cancelled, nil
		}
		return acpsdk.PromptResponse{}, internalError("执行 Agent", err)
	}
	for ev := range events {
		switch ev.Type {
		case "text":
			if ev.Delta == "" {
				continue
			}
			if err := s.update(ctx, acpsdk.UpdateAgentMessageText(ev.Delta)); err != nil && ctx.Err() == nil {
				return acpsdk.PromptResponse{}, internalError("发送回答", err)
			}
		case "error":
			if ctx.Err() != nil {
				return cancelled, nil
			}
			cause := ev.Error
			if cause == nil {
				cause = errors.New(ev.Message)
			}
			return acpsdk.PromptResponse{}, internalError("执行 Agent", cause)
		case "done":
			return acpsdk.PromptResponse{StopReason: acpsdk.StopReasonEndTurn}, nil
		}
	}
	if ctx.Err() != nil {
		return cancelled, nil
	}
	return acpsdk.PromptResponse{}, internalError("执行 Agent", errors.New("事件流在 done 之前结束"))
}

// promptText 把内容块拼成 Agent 输入：文本原样，资源链接只写 URI，嵌入的文本资源以 <uri> 加内容附上。
func promptText(blocks []acpsdk.ContentBlock) (string, error) {
	var b strings.Builder
	for i, blk := range blocks {
		if i > 0 {
			b.WriteString("\n\n")
		}
		switch {
		case blk.Text != nil:
			b.WriteString(blk.Text.Text)
		case blk.ResourceLink != nil:
			b.WriteString(blk.ResourceLink.Uri)
		case blk.Resource != nil && blk.Resource.Resource.TextResourceContents != nil:
			r := blk.Resource.Resource.TextResourceContents
			fmt.Fprintf(&b, "<%s>\n%s", r.Uri, r.Text)
		default:
			return "", errors.New("acp: 不支持的内容块，只支持 text、resource_link 与文本 resource")
		}
	}
	return b.String(), nil
}
```

```go
// 追加到 harness_test.go；import 增加 "fmt"、"github.com/kart-io/goagent/core/session"、"github.com/kart-io/goagent/tools"。

// toolFactory 构建经 s.Pipeline() 注册工具的 Agent；ask 非 nil 时作为宿主的审批钩子追加，store 非 nil 时开启会话日志。
func toolFactory(model llm.Client, ask tools.PreHook, store session.Store, build func(*Session) []interfaces.Tool) Factory {
	return func(_ context.Context, s *Session) (Runner, error) {
		p := s.Pipeline()
		if ask != nil {
			p.Pre = append(p.Pre, ask)
		}
		reg := tools.NewRegistry()
		for _, tool := range build(s) {
			if err := reg.Register(tool); err != nil {
				return nil, err
			}
		}
		d, err := tools.NewDispatcher(reg, tools.DispatchConfig{Pipeline: p})
		if err != nil {
			return nil, err
		}
		b := builder.NewSimpleBuilder(model).WithSessionID(s.ID()).WithTools(d.Tools()...).WithMaxIterations(4)
		if store != nil {
			b = b.WithSessionLog(store)
		}
		agent, err := b.Build()
		if err != nil {
			return nil, err
		}
		return agent, nil
	}
}
```

```go
package acp

import (
	"context"
	"encoding/json"
	"fmt"
	"sync"
	"time"

	acpsdk "github.com/coder/acp-go-sdk"
	"github.com/kart-io/goagent/interfaces"
	"github.com/kart-io/goagent/tools"
)

const (
	optionAllowOnce    = "allow_once"
	optionAllowSession = "allow_always"
	optionReject       = "reject_once"
)

// endTurnTimeout 限制一轮结束时补发通知的等待；此时 prompt 的 ctx 可能已取消。
const endTurnTimeout = 5 * time.Second

// Pipeline 返回已装好工具调用通知与审批的管线，交给宿主的 tools.Dispatcher。宿主追加返回 PreAsk 的
// PreHook 来要求审批，也可追加 Guards、Around、Post，但不应替换第 0 项与 Approver。
func (s *Session) Pipeline() *tools.Pipeline { return s.pipeline }

func (s *Session) newPipeline() *tools.Pipeline {
	return &tools.Pipeline{
		Pre:      []tools.PreHook{s.announce},
		Approver: tools.ApproverFunc(s.approve),
		Around:   []tools.Around{s.running},
		Post:     []tools.PostHook{s.finished},
	}
}

func (s *Session) check(call *tools.Call) error {
	if call.SessionID != s.id {
		return fmt.Errorf("acp: 工具调用属于会话 %q 而不是 ACP 会话 %q，构建 Agent 时须 WithSessionID(s.ID())", call.SessionID, s.id)
	}
	return nil
}

func (s *Session) announce(ctx context.Context, call *tools.Call) (tools.PreDecision, error) {
	if err := s.check(call); err != nil {
		return tools.PreDecision{}, err
	}
	id := acpsdk.ToolCallId(call.CallID)
	kind := s.conn.srv.kind(call.Tool)
	// 替代调用以同一 CallID 再次经过管线，只更新标题与参数。
	if !s.calls.start(call.CallID) {
		return tools.PreDecision{}, s.update(ctx, acpsdk.UpdateToolCall(id, acpsdk.WithUpdateTitle(call.Tool), acpsdk.WithUpdateKind(kind), acpsdk.WithUpdateRawInput(call.Args)))
	}
	return tools.PreDecision{}, s.update(ctx, acpsdk.StartToolCall(id, call.Tool,
		acpsdk.WithStartKind(kind), acpsdk.WithStartStatus(acpsdk.ToolCallStatusPending), acpsdk.WithStartRawInput(call.Args)))
}

func (s *Session) approve(ctx context.Context, req tools.ApprovalRequest) (tools.ApprovalResult, error) {
	options := []acpsdk.PermissionOption{{Kind: acpsdk.PermissionOptionKindAllowOnce, Name: "允许一次", OptionId: optionAllowOnce}}
	if req.Grant != nil {
		options = append(options, acpsdk.PermissionOption{Kind: acpsdk.PermissionOptionKindAllowAlways, Name: "本会话允许：" + req.Grant.Describe(), OptionId: optionAllowSession})
	}
	options = append(options, acpsdk.PermissionOption{Kind: acpsdk.PermissionOptionKindRejectOnce, Name: "拒绝", OptionId: optionReject})
	title := req.Call.Tool
	if req.Reason != "" {
		title += "：" + req.Reason
	}
	resp, err := s.conn.client().RequestPermission(ctx, acpsdk.RequestPermissionRequest{
		SessionId: acpsdk.SessionId(s.id),
		ToolCall: acpsdk.ToolCallUpdate{
			ToolCallId: acpsdk.ToolCallId(req.Call.CallID),
			Title:      acpsdk.Ptr(title),
			Status:     acpsdk.Ptr(acpsdk.ToolCallStatusPending),
			RawInput:   req.Call.Args,
		},
		Options: options,
	})
	if err != nil {
		if ctx.Err() != nil {
			return tools.ApprovalResult{Outcome: tools.ApprovalCancelled}, nil
		}
		return tools.ApprovalResult{}, fmt.Errorf("acp: 请求审批：%w", err)
	}
	if resp.Outcome.Selected == nil {
		return tools.ApprovalResult{Outcome: tools.ApprovalCancelled}, nil
	}
	switch string(resp.Outcome.Selected.OptionId) {
	case optionAllowOnce:
		return tools.ApprovalResult{Outcome: tools.ApprovalAllowedOnce}, nil
	case optionAllowSession:
		return tools.ApprovalResult{Outcome: tools.ApprovalAllowedSession}, nil
	default:
		return tools.ApprovalResult{Outcome: tools.ApprovalRejected}, nil
	}
}

func (s *Session) running(ctx context.Context, call *tools.Call, next func(context.Context) (*interfaces.ToolOutput, error)) (*interfaces.ToolOutput, error) {
	id := acpsdk.ToolCallId(call.CallID)
	if err := s.update(ctx, acpsdk.UpdateToolCall(id, acpsdk.WithUpdateStatus(acpsdk.ToolCallStatusInProgress))); err != nil {
		return nil, err
	}
	out, err := next(withCallID(ctx, call.CallID))
	if err != nil {
		s.fail(ctx, call.CallID, err.Error())
	}
	return out, err
}

func (s *Session) finished(ctx context.Context, call *tools.Call, out *interfaces.ToolOutput) (tools.PostDecision, error) {
	id := acpsdk.ToolCallId(call.CallID)
	terminals := s.calls.finish(call.CallID)
	if msg, rejected := tools.RejectedResult(out); rejected {
		return tools.PostDecision{}, s.settle(ctx, acpsdk.UpdateToolCall(id,
			acpsdk.WithUpdateStatus(acpsdk.ToolCallStatusFailed), acpsdk.WithUpdateContent(toolContent(terminals, msg))))
	}
	return tools.PostDecision{}, s.settle(ctx, acpsdk.UpdateToolCall(id,
		acpsdk.WithUpdateStatus(acpsdk.ToolCallStatusCompleted),
		acpsdk.WithUpdateContent(toolContent(terminals, resultText(out))),
		acpsdk.WithUpdateRawOutput(out.Result)))
}

// fail 发送 failed，发送失败不影响调用方的返回值。
func (s *Session) fail(ctx context.Context, callID, text string) {
	terminals := s.calls.finish(callID)
	_ = s.settle(ctx, acpsdk.UpdateToolCall(acpsdk.ToolCallId(callID),
		acpsdk.WithUpdateStatus(acpsdk.ToolCallStatusFailed), acpsdk.WithUpdateContent(toolContent(terminals, text))))
}

// settle 发送调用的终态通知。调用已从未结束表移除，ctx 此时可能已取消（如审批等待中取消 prompt），
// 发送失败就再无补发机会，所以改用不随之取消的有界 ctx。
func (s *Session) settle(ctx context.Context, u acpsdk.SessionUpdate) error {
	ctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), endTurnTimeout)
	defer cancel()
	return s.update(ctx, u)
}

// endTurn 为本轮已发出 tool_call 但未结束的调用补发 failed（准入阶段出错、终止或取消时 Around 与 Post 都不运行）。
func (s *Session) endTurn(ctx context.Context) {
	for _, id := range s.calls.drain() {
		s.fail(ctx, id, "调用未完成")
	}
}

func toolContent(terminals []string, text string) []acpsdk.ToolCallContent {
	var out []acpsdk.ToolCallContent
	for _, t := range terminals {
		out = append(out, acpsdk.ToolTerminalRef(t))
	}
	if text != "" {
		out = append(out, acpsdk.ToolContent(acpsdk.TextBlock(text)))
	}
	return out
}

// resultText 把工具结果转为编辑器显示的文本：字符串原样，其他类型编码为 JSON。
func resultText(out *interfaces.ToolOutput) string {
	switch v := out.Result.(type) {
	case nil:
		return ""
	case string:
		return v
	default:
		data, err := json.Marshal(v)
		if err != nil {
			return fmt.Sprint(v)
		}
		return string(data)
	}
}

// callTracker 记录已发出 tool_call 但未结束的调用及其终端。
type callTracker struct {
	mu   sync.Mutex
	open map[string][]string
}

func newCallTracker() *callTracker { return &callTracker{open: map[string][]string{}} }

func (t *callTracker) start(id string) bool {
	t.mu.Lock()
	defer t.mu.Unlock()
	if _, ok := t.open[id]; ok {
		return false
	}
	t.open[id] = nil
	return true
}

func (t *callTracker) addTerminal(id, term string) []string {
	t.mu.Lock()
	defer t.mu.Unlock()
	t.open[id] = append(t.open[id], term)
	return append([]string(nil), t.open[id]...)
}

func (t *callTracker) finish(id string) []string {
	t.mu.Lock()
	defer t.mu.Unlock()
	terms := t.open[id]
	delete(t.open, id)
	return terms
}

func (t *callTracker) drain() []string {
	t.mu.Lock()
	defer t.mu.Unlock()
	ids := make([]string, 0, len(t.open))
	for id := range t.open {
		ids = append(ids, id)
	}
	return ids
}

type callIDKey struct{}

func withCallID(ctx context.Context, id string) context.Context {
	return context.WithValue(ctx, callIDKey{}, id)
}

func callIDFrom(ctx context.Context) string {
	id, _ := ctx.Value(callIDKey{}).(string)
	return id
}
```

```go
import (
	"context"
```

```go
// LoadSession 回放会话日志中的对话，再以同一 ID 构建 Agent 续接。
func (c *conn) LoadSession(ctx context.Context, p acpsdk.LoadSessionRequest) (acpsdk.LoadSessionResponse, error) {
	if c.srv.store == nil {
		return acpsdk.LoadSessionResponse{}, acpsdk.NewMethodNotFound(acpsdk.AgentMethodSessionLoad)
	}
	id := string(p.SessionId)
	events, err := c.srv.store.Read(ctx, id, 0)
	if errors.Is(err, session.ErrNotFound) {
		return acpsdk.LoadSessionResponse{}, acpsdk.NewInvalidParams(map[string]any{"error": fmt.Sprintf("acp: 会话 %q 不存在", id)})
	}
	if err != nil {
		return acpsdk.LoadSessionResponse{}, internalError("读取会话日志", err)
	}
	s, err := c.open(ctx, id, p.Cwd, p.McpServers)
	if err != nil {
		return acpsdk.LoadSessionResponse{}, err
	}
	if err := s.replay(ctx, events); err != nil {
		return acpsdk.LoadSessionResponse{}, internalError("回放会话", err)
	}
	return acpsdk.LoadSessionResponse{}, nil
}

// replay 按 session.Derive 推导的顺序发送历史；被退回与中止轮次中的消息已被 Derive 排除。
func (s *Session) replay(ctx context.Context, events []session.Event) error {
	entries, err := session.Derive(events)
	if err != nil {
		return err
	}
	for _, e := range entries {
		m := e.Message
		var updates []acpsdk.SessionUpdate
		switch m.Role {
		case "user":
			updates = append(updates, acpsdk.UpdateUserMessageText(m.Content))
		case "assistant":
			if m.Content != "" {
				updates = append(updates, acpsdk.UpdateAgentMessageText(m.Content))
			}
			for _, tc := range m.ToolCalls {
				var args any
				if json.Unmarshal([]byte(tc.Function.Arguments), &args) != nil {
					args = tc.Function.Arguments
				}
				updates = append(updates, acpsdk.StartToolCall(acpsdk.ToolCallId(tc.ID), tc.Function.Name,
					acpsdk.WithStartKind(s.conn.srv.kind(tc.Function.Name)),
					acpsdk.WithStartStatus(acpsdk.ToolCallStatusPending), acpsdk.WithStartRawInput(args)))
			}
		case "tool":
			status := acpsdk.ToolCallStatusCompleted
			if e.Tool != nil && !e.Tool.Success {
				status = acpsdk.ToolCallStatusFailed
			}
			updates = append(updates, acpsdk.UpdateToolCall(acpsdk.ToolCallId(m.ToolCallID),
				acpsdk.WithUpdateStatus(status), acpsdk.WithUpdateContent(toolContent(nil, m.Content))))
		}
		for _, u := range updates {
			if err := s.update(ctx, u); err != nil {
				return err
			}
		}
	}
	return nil
}

// ListSessions 列出会话日志中的会话；工作区只对本连接新建或加载过的会话已知，未知时为空且不参与 cwd 过滤。
func (c *conn) ListSessions(ctx context.Context, p acpsdk.ListSessionsRequest) (acpsdk.ListSessionsResponse, error) {
	if c.srv.store == nil {
		return acpsdk.ListSessionsResponse{}, acpsdk.NewMethodNotFound(acpsdk.AgentMethodSessionList)
	}
	ids, err := c.srv.store.List(ctx)
	if err != nil {
		return acpsdk.ListSessionsResponse{}, internalError("列出会话", err)
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	out := acpsdk.ListSessionsResponse{Sessions: []acpsdk.SessionInfo{}}
	for _, id := range ids {
		cwd, known := c.cwds[id]
		if p.Cwd != nil && known && cwd != *p.Cwd {
			continue
		}
		out.Sessions = append(out.Sessions, acpsdk.SessionInfo{SessionId: acpsdk.SessionId(id), Cwd: cwd})
	}
	return out, nil
}
```

```go
package acp

import (
	"context"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"fmt"
	"io/fs"
	"path/filepath"

	acpsdk "github.com/coder/acp-go-sdk"
	"github.com/kart-io/goagent/harness/env"
)

// clientFS 把读写转发给编辑器，使 Agent 读到未保存的缓冲区、写入出现在编辑器中；
// 路径限定与 ACP 没有的操作（Stat、Walk、Remove）沿用工作区的 LocalFS。
// 编辑器不参与写入协调，条件写入的检查与写入之间的修改无法发现。
type clientFS struct {
	s           *Session
	read, write bool
}

var _ env.FileSystem = (*clientFS)(nil)

func (f *clientFS) Resolve(path string) (string, error) { return f.s.local.Resolve(path) }

func (f *clientFS) Stat(ctx context.Context, path string) (env.Info, error) {
	return f.s.local.Stat(ctx, path)
}

func (f *clientFS) Remove(ctx context.Context, path string) error { return f.s.local.Remove(ctx, path) }

func (f *clientFS) Walk(ctx context.Context, path string, fn func(env.Info) error) error {
	return f.s.local.Walk(ctx, path, fn)
}

// abs 把工作区路径转为 ACP 要求的绝对路径。
func (f *clientFS) abs(path string) (string, error) {
	rel, err := f.s.local.Resolve(path)
	if err != nil {
		return "", err
	}
	return filepath.Join(f.s.cwd, filepath.FromSlash(rel)), nil
}

func (f *clientFS) ReadFile(ctx context.Context, path string) ([]byte, env.Version, error) {
	if !f.read {
		return f.s.local.ReadFile(ctx, path)
	}
	abs, err := f.abs(path)
	if err != nil {
		return nil, "", err
	}
	// ACP 没有标准的"不存在"错误码，存在性以磁盘为准。
	if _, err := f.s.local.Stat(ctx, path); err != nil {
		return nil, "", err
	}
	resp, err := f.s.conn.client().ReadTextFile(ctx, acpsdk.ReadTextFileRequest{SessionId: acpsdk.SessionId(f.s.id), Path: abs})
	if err != nil {
		return nil, "", fmt.Errorf("acp: 经编辑器读取 %s：%w", path, err)
	}
	data := []byte(resp.Content)
	return data, versionOf(data), nil
}

func (f *clientFS) WriteFile(ctx context.Context, path string, data []byte, intent env.WriteIntent) (env.Version, error) {
	if !f.write {
		return f.s.local.WriteFile(ctx, path, data, intent)
	}
	abs, err := f.abs(path)
	if err != nil {
		return "", err
	}
	if intent.CreateIfAbsent {
		_, err := f.s.local.Stat(ctx, path)
		if err == nil {
			return "", env.ErrExists
		}
		if !errors.Is(err, fs.ErrNotExist) {
			return "", err
		}
	}
	if intent.IfVersion != "" {
		_, current, err := f.ReadFile(ctx, path)
		if err != nil {
			return "", err
		}
		if current != intent.IfVersion {
			return "", env.ErrStale
		}
	}
	if _, err := f.s.conn.client().WriteTextFile(ctx, acpsdk.WriteTextFileRequest{SessionId: acpsdk.SessionId(f.s.id), Path: abs, Content: string(data)}); err != nil {
		return "", fmt.Errorf("acp: 经编辑器写入 %s：%w", path, err)
	}
	return versionOf(data), nil
}

// versionOf 与 env.LocalFS 的版本格式一致：内容 SHA-256 的十六进制。
func versionOf(data []byte) env.Version {
	sum := sha256.Sum256(data)
	return env.Version(hex.EncodeToString(sum[:]))
}
```

```go
package acp

import (
	"context"
	"errors"
	"fmt"
	"path/filepath"

	acpsdk "github.com/coder/acp-go-sdk"
	"github.com/kart-io/goagent/harness/env"
)

// clientExecutor 在编辑器终端中执行命令，编辑器可展示输出；工具调用内执行时终端嵌入该调用。
type clientExecutor struct{ s *Session }

var _ env.Executor = (*clientExecutor)(nil)

// Capabilities 实现 env.Executor：编辑器终端在宿主机上执行，可以访问网络。
func (e *clientExecutor) Capabilities() env.Capabilities { return env.Capabilities{Network: true} }

func (e *clientExecutor) Run(ctx context.Context, cmd env.Command) (env.Result, error) {
	if len(cmd.Argv) == 0 {
		return env.Result{}, errors.New("acp: 命令为空")
	}
	rel, err := e.s.local.Resolve(cmd.Dir)
	if err != nil {
		return env.Result{}, err
	}
	dir := filepath.Join(e.s.cwd, filepath.FromSlash(rel))
	peer := e.s.conn.client()
	sid := acpsdk.SessionId(e.s.id)
	created, err := peer.CreateTerminal(ctx, acpsdk.CreateTerminalRequest{SessionId: sid, Command: cmd.Argv[0], Args: cmd.Argv[1:], Cwd: &dir})
	if err != nil {
		return env.Result{}, fmt.Errorf("acp: 创建终端：%w", err)
	}
	term := created.TerminalId
	// 结束阶段的请求不能随调用方 ctx 取消，否则终端得不到 kill 与 release。
	cleanup, cancel := context.WithTimeout(context.WithoutCancel(ctx), endTurnTimeout)
	defer cancel()
	defer func() { _, _ = peer.ReleaseTerminal(cleanup, acpsdk.ReleaseTerminalRequest{SessionId: sid, TerminalId: term}) }()

	if callID := callIDFrom(ctx); callID != "" {
		terms := e.s.calls.addTerminal(callID, term)
		if err := e.s.update(ctx, acpsdk.UpdateToolCall(acpsdk.ToolCallId(callID), acpsdk.WithUpdateContent(toolContent(terms, "")))); err != nil {
			return env.Result{}, err
		}
	}

	exit, err := peer.WaitForTerminalExit(ctx, acpsdk.WaitForTerminalExitRequest{SessionId: sid, TerminalId: term})
	if err != nil {
		if ctx.Err() == nil {
			return env.Result{}, fmt.Errorf("acp: 等待终端退出：%w", err)
		}
		_, _ = peer.KillTerminal(cleanup, acpsdk.KillTerminalRequest{SessionId: sid, TerminalId: term})
		out, _ := peer.TerminalOutput(cleanup, acpsdk.TerminalOutputRequest{SessionId: sid, TerminalId: term})
		return env.Result{Output: []byte(out.Output)}, ctx.Err()
	}
	out, err := peer.TerminalOutput(ctx, acpsdk.TerminalOutputRequest{SessionId: sid, TerminalId: term})
	if err != nil {
		return env.Result{}, fmt.Errorf("acp: 读取终端输出：%w", err)
	}
	code := 0
	switch {
	case exit.ExitCode != nil:
		code = *exit.ExitCode
	case exit.Signal != nil:
		code = -1 // 被信号终止，没有退出码
	}
	return env.Result{Output: []byte(out.Output), ExitCode: code}, nil
}
```

````markdown
# acp — 以 ACP Agent 身份运行 goagent

[Agent Client Protocol](https://agentclientprotocol.com)（ACP）是编辑器与编码 Agent 之间的 stdio JSON-RPC 协议。本模块把任意 `builder` 构建的 Agent 暴露为 ACP Agent，供 Zed、JetBrains、Neovim 等编辑器驱动。协议实现基于 [coder/acp-go-sdk](https://github.com/coder/acp-go-sdk) v0.13.5（协议版本 1）。

```go
srv := acp.NewServer(func(ctx context.Context, s *acp.Session) (acp.Runner, error) {
    p := s.Pipeline()
    p.Pre = append(p.Pre, askBeforeWrite) // 返回 tools.PreAsk 的钩子触发编辑器审批
    reg := tools.NewRegistry()
    _ = reg.Register(readFileTool(s.FileSystem()))
    _ = reg.Register(runCommandTool(s.Executor()))
    d, err := tools.NewDispatcher(reg, tools.DispatchConfig{Pipeline: p})
    if err != nil {
        return nil, err
    }
    return builder.NewSimpleBuilder(model).
        WithSessionID(s.ID()).   // 必须：工具通知按会话 ID 路由
        WithSessionLog(store).   // 与 WithSessionStore 同一个 Store
        WithTools(d.Tools()...).
        Build()
}, acp.WithSessionStore(store), acp.WithAgentInfo("my-agent", "1.0.0"))
log.SetOutput(os.Stderr) // stdout 被协议占用
if err := srv.Serve(ctx, os.Stdin, os.Stdout); err != nil {
    log.Fatal(err)
}
```

## 装配约定

- 每个 ACP 会话调用一次 Factory，构建时必须 `WithSessionID(s.ID())`，否则工具调用报错。
- 工具须经 `s.Pipeline()` 所在的 `tools.Dispatcher` 注册，才有工具通知与审批；本模块无法检测遗漏。
- 审批：宿主向 `s.Pipeline().Pre` 追加返回 `tools.PreAsk` 的钩子；带 `Grant` 时编辑器多一个"本会话允许"。不追加则直接执行。不要替换已装好的第 0 项与 `Approver`；宿主追加的 Post 替换的结果不反映在编辑器通知中。
- `WithSessionStore` 开启 `session/load` 与 `session/list`，Factory 须用同一 Store 调用 `WithSessionLog`（会话日志只支持原生工具流）。
- `s.MCPServers()` 是编辑器要求连接的 MCP 服务器，可交给 goagent 的 `mcp/remote`。

## 方法支持

| ACP | 支持 |
|---|---|
| `initialize`、`session/new`、`session/prompt`、`session/cancel`、`session/close` | 是 |
| `session/load`、`session/list` | 配置 `WithSessionStore` 时 |
| `authenticate` | 不声明认证方式，返回空响应 |
| `logout`、`session/set_mode`、`session/set_config_option`、`session/resume` | method-not-found |
| prompt 内容 | `text`、`resource_link`、文本 `resource`；图片与音频返回 invalid-params |

## 文件与命令

- `s.FileSystem()`：客户端声明 fs 能力时，`ReadFile`/`WriteFile` 经编辑器（可读到未保存的缓冲区，写入出现在编辑器中），存在性、`Stat`、`Walk`、`Remove` 与路径限定使用以工作区为根的 `env.LocalFS`。编辑器不参与写入协调，`CreateIfAbsent`、`IfVersion` 只是尽力而为；只存在于编辑器、未落盘的新缓冲区读不到。
- `s.Executor()`：客户端声明 terminal 能力时命令在编辑器终端中执行，工具调用内执行的终端嵌入该调用；ctx 结束时 kill 并 release。被信号终止的命令 `ExitCode` 为 -1。否则使用 `env.LocalExecutor`。

{
  "agent_servers": {
    "my-agent": { "command": "/path/to/my-agent", "args": [], "env": {} }
  }
}
```
````

```go
// ACP Agent：以 Agent Client Protocol 经 stdio 运行，可在 Zed 等编辑器中使用。
// 提供读文件与执行命令两个工具，执行命令前经编辑器审批。默认使用离线模型；GOAGENT_LIVE=1 时使用真实模型。
package main

import (
	"context"
	"fmt"
	"os"
	"strings"
	"sync/atomic"

	acpsdk "github.com/coder/acp-go-sdk"
	"github.com/kart-io/goagent-contrib/acp"

	"github.com/kart-io/goagent-examples/orchestration/internal/demomodel"

// demoModel 先读 README，再执行 ls，最后用读到的内容作答。
func demoModel() llm.ToolStreamClient {
	var calls atomic.Int64
	return &demomodel.Model{Calls: &calls, Respond: func(r *llm.CompletionRequest) llm.StreamChunk {
		var results []string
		for _, m := range r.Messages {
			if m.Role == "tool" {
				results = append(results, m.Content)
			}
		}
		switch len(results) {
		case 0:
			c := demomodel.Call("c1", "read_file")
			c.Function.Arguments = `{"path":"README.md"}`
			return llm.StreamChunk{ToolCalls: []llm.ToolCall{c}}
		case 1:
			c := demomodel.Call("c2", "run_command")
			c.Function.Arguments = `{"argv":["ls"]}`
			return llm.StreamChunk{ToolCalls: []llm.ToolCall{c}}
		default:
			answer := "项目说明：" + strings.TrimSpace(results[0])
			return llm.StreamChunk{Content: answer, Delta: answer}
		}
	}}
}

func readFile(fs env.FileSystem) interfaces.Tool {
	return tools.NewBaseTool("read_file", "读取工作区内的文本文件", `{"type":"object","properties":{"path":{"type":"string"}},"required":["path"]}`,
		func(ctx context.Context, in *interfaces.ToolInput) (*interfaces.ToolOutput, error) {
			path, _ := in.Args["path"].(string)
			data, _, err := fs.ReadFile(ctx, path)
			if err != nil {
				return tools.Rejected(err.Error(), nil), nil
			}
			return &interfaces.ToolOutput{Success: true, Result: string(data)}, nil
		})
}

func runCommand(exec env.Executor) interfaces.Tool {
	return tools.NewBaseTool("run_command", "在工作区内执行命令（不经 shell）", `{"type":"object","properties":{"argv":{"type":"array","items":{"type":"string"}}},"required":["argv"]}`,
		func(ctx context.Context, in *interfaces.ToolInput) (*interfaces.ToolOutput, error) {
			raw, _ := in.Args["argv"].([]any)
			argv := make([]string, 0, len(raw))
			for _, a := range raw {
				argv = append(argv, fmt.Sprint(a))
			}
			res, err := exec.Run(ctx, env.Command{Argv: argv})
			if err != nil {
				return nil, err
			}
			return &interfaces.ToolOutput{Success: true, Result: fmt.Sprintf("退出码 %d\n%s", res.ExitCode, res.Output)}, nil
		})
}

// askBeforeCommand 要求执行命令前审批，本会话允许按命令名放行。
func askBeforeCommand(_ context.Context, c *tools.Call) (tools.PreDecision, error) {
	if c.Tool != "run_command" {
		return tools.PreDecision{}, nil
	}
	return tools.PreDecision{Kind: tools.PreAsk, Feedback: "将在工作区执行命令", Grant: commandGrant{}}, nil
}

type commandGrant struct{}

func (commandGrant) Describe() string           { return "本会话内的全部命令" }
func (commandGrant) Matches(c *tools.Call) bool { return c.Tool == "run_command" }

func newServer(model llm.ToolStreamClient) *acp.Server {
	kinds := map[string]acpsdk.ToolKind{"read_file": acpsdk.ToolKindRead, "run_command": acpsdk.ToolKindExecute}
	return acp.NewServer(func(_ context.Context, s *acp.Session) (acp.Runner, error) {
		p := s.Pipeline()
		p.Pre = append(p.Pre, askBeforeCommand)
		reg := tools.NewRegistry()
		for _, t := range []interfaces.Tool{readFile(s.FileSystem()), runCommand(s.Executor())} {
			if err := reg.Register(t); err != nil {
				return nil, err
			}
		}
		d, err := tools.NewDispatcher(reg, tools.DispatchConfig{Pipeline: p})
		if err != nil {
			return nil, err
		}
		agent, err := builder.NewSimpleBuilder(model).
			WithSystemPrompt("你是编码助手，工作区为 " + s.Cwd() + "。").
			WithSessionID(s.ID()).
			WithTools(d.Tools()...).
			WithMaxIterations(6).
			Build()
		if err != nil {
			return nil, err
		}
		return agent, nil
	}, acp.WithAgentInfo("goagent-acp-example", "0.1.0"), acp.WithToolKind(func(tool string) acpsdk.ToolKind {
		if k, ok := kinds[tool]; ok {
			return k
		}
		return acpsdk.ToolKindOther
	}))
}

func main() {
	var model llm.ToolStreamClient = demoModel()
	if os.Getenv("GOAGENT_LIVE") == "1" {
		client, err := livemodel.FromEnv()
		if err != nil {
			fmt.Fprintln(os.Stderr, err)
			os.Exit(1)
		}
		model = client
	}
	if err := newServer(model).Serve(context.Background(), os.Stdin, os.Stdout); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
```
